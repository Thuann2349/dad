    <script>
        let files = {
            "index.html": `<!DOCTYPE html>\n<html lang="pt-BR">\n<head>\n    <meta charset="UTF-8">\n    <script src="https://cdn.tailwindcss.com"><\/script>\n</head>\n<body class="bg-slate-50 flex items-center justify-center h-screen">\n    <div class="text-center p-8 bg-white shadow-xl rounded-2xl border border-slate-100">\n        <h1 class="text-2xl font-bold text-indigo-600 mb-2">Projeto Pronto!</h1>\n        <p class="text-slate-500 text-sm">Descreva algo no chat para gerar a aplicação.</p>\n    </div>\n</body>\n</html>`
        };
        let currentFile = "index.html";

        function openSettings() {
            document.getElementById('api-key-input').value = localStorage.getItem('useoneai_key') || '';
            document.getElementById('model-select').value = localStorage.getItem('useoneai_model') || 'chatgpt-5.6-terra';
            document.getElementById('settings-modal').classList.remove('hidden');
        }

        function closeSettings() {
            document.getElementById('settings-modal').classList.add('hidden');
        }

        function saveSettings() {
            const key = document.getElementById('api-key-input').value.trim();
            const model = document.getElementById('model-select').value;
            localStorage.setItem('useoneai_key', key);
            localStorage.setItem('useoneai_model', model);
            closeSettings();
            alert('Configurações salvas com sucesso!');
        }

        function renderFilesList() {
            const list = document.getElementById('file-list');
            list.innerHTML = '';
            for (let filename in files) {
                const item = document.div || document.createElement('div');
                item.className = `px-3 py-2 rounded cursor-pointer text-xs flex items-center justify-between ${filename === currentFile ? 'bg-indigo-600/20 text-indigo-300 font-semibold border border-indigo-500/30' : 'hover:bg-slate-800 text-slate-300'}`;
                item.innerHTML = `<span class="flex items-center gap-2"><i class="fa-regular fa-file-code"></i> ${filename}</span>`;
                item.onclick = () => { currentFile = filename; refreshPreview(); renderFilesList(); };
                list.appendChild(item);
            }
        }

        function refreshPreview() {
            const frame = document.getElementById('preview-frame');
            frame.srcdoc = files["index.html"] || '';
        }

        async function generateProject() {
            const prompt = document.getElementById('user-prompt').value.trim();
            const apiKey = localStorage.getItem('useoneai_key');
            const model = localStorage.getItem('useoneai_model') || 'chatgpt-5.6-terra';

            if (!apiKey) {
                alert('Por favor, configure sua chave de API nas configurações primeiro.');
                openSettings();
                return;
            }
            if (!prompt) return;

            const chatMessages = document.getElementById('chat-messages');
            chatMessages.innerHTML += `<div class="bg-indigo-600/20 border border-indigo-500/30 p-3 rounded-lg text-slate-200"><strong>Você:</strong> ${prompt}</div>`;
            document.getElementById('user-prompt').value = '';

            chatMessages.innerHTML += `<div id="loading-msg" class="bg-slate-800 p-3 rounded-lg text-slate-400 animate-pulse">Gerando código com IA...</div>`;
            chatMessages.scrollTop = chatMessages.scrollHeight;

            try {
                const response = await fetch('https://api.useoneai.app/v1/chat/completions', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'Authorization': `Bearer ${apiKey}`
                    },
                    body: JSON.stringify({
                        model: model,
                        messages: [
                            { role: "system", content: "Você é um gerador de aplicações web completas em código HTML único contendo Tailwind CSS integrado. Retorne estritamente apenas o código HTML completo da aplicação solicitada, sem textos adicionais ou blocos markdown de explicação." },
                            { role: "user", content: prompt }
                        ]
                    })
                });

                const data = await response.json();
                document.getElementById('loading-msg').remove();

                if (data.choices && data.choices[0]) {
                    let code = data.choices[0].message.content;
                    // Remove marcações markdown extras caso a IA retorne
                    code = code.replace(/```html/g, '').replace(/```/g, '').trim();
                    
                    files["index.html"] = code;
                    renderFilesList();
                    refreshPreview();

                    chatMessages.innerHTML += `<div class="bg-emerald-600/20 border border-emerald-500/30 p-3 rounded-lg text-emerald-200">✨ Aplicação gerada com sucesso!</div>`;
                } else {
                    throw new Error('Resposta inválida da API');
                }
            } catch (err) {
                if (document.getElementById('loading-msg')) document.getElementById('loading-msg').remove();
                chatMessages.innerHTML += `<div class="bg-red-600/20 border border-red-500/30 p-3 rounded-lg text-red-200">Erro ao gerar: ${err.message}</div>`;
            }
            chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        // Inicialização
        renderFilesList();
        refreshPreview();
    </script>
